from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import anthropic
import os
from supabase import create_client
import voyageai
import httpx

app = FastAPI()

# Clients
supabase = create_client(os.environ["SUPABASE_URL"], os.environ["SUPABASE_KEY"])
claude = anthropic.Anthropic(api_key=os.environ["ANTHROPIC_API_KEY"])
voyage = voyageai.Client(api_key=os.environ["VOYAGE_API_KEY"])

REDMINE_URL = os.environ["REDMINE_URL"]
REDMINE_API_KEY = os.environ["REDMINE_API_KEY"]


class Issue(BaseModel):
    id: int
    title: str
    description: str


def get_embedding(text: str) -> list:
    result = voyage.embed([text], model="voyage-2")
    return result.embeddings[0]


def find_similar_issues(embedding: list, limit: int = 5) -> list:
    result = supabase.rpc("match_issues", {
        "query_embedding": embedding,
        "match_count": limit
    }).execute()
    return result.data or []


def store_issue(issue: Issue, embedding: list):
    supabase.table("issues").upsert({
        "redmine_id": issue.id,
        "title": issue.title,
        "description": issue.description,
        "embedding": embedding
    }).execute()


async def post_redmine_note(issue_id: int, note: str):
    url = f"{REDMINE_URL}/issues/{issue_id}.json"
    headers = {"X-Redmine-API-Key": REDMINE_API_KEY, "Content-Type": "application/json"}
    payload = {"issue": {"notes": note}}
    async with httpx.AsyncClient() as client:
        await client.put(url, json=payload, headers=headers)


@app.post("/analyze")
async def analyze_issue(issue: Issue):
    try:
        # 1. Embed the new issue
        embedding = get_embedding(f"{issue.title}\n{issue.description}")

        # 2. Find similar past issues
        similar = find_similar_issues(embedding)

        # 3. Build context from similar issues
        context = ""
        if similar:
            context = "Similar past issues:\n\n"
            for s in similar:
                context += f"- [{s['redmine_id']}] {s['title']}: {s['description'][:300]}\n\n"
        else:
            context = "No similar past issues found.\n"

        # 4. Ask Claude to analyze
        prompt = f"""You are an expert software project assistant with access to historical issue data.

A new issue has been created in Redmine. Analyze it and:
1. Identify if it's similar to any past issues
2. Suggest a likely solution or next steps based on history
3. Highlight any patterns or recurring problems you notice

New Issue:
Title: {issue.title}
Description: {issue.description}

{context}

Provide a concise, actionable analysis."""

        response = claude.messages.create(
            model="claude-sonnet-4-6",
            max_tokens=1000,
            messages=[{"role": "user", "content": prompt}]
        )

        analysis = response.content[0].text

        # 5. Store this issue for future reference
        store_issue(issue, embedding)

        # 6. Post analysis back to Redmine
        note = f"🤖 **Claude AI Analysis**\n\n{analysis}"
        await post_redmine_note(issue.id, note)

        return {"status": "ok", "analysis": analysis}

    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@app.post("/sync")
async def sync_all_issues():
    """Call this once to import all existing Redmine issues into the vector DB."""
    headers = {"X-Redmine-API-Key": REDMINE_API_KEY}
    offset = 0
    limit = 100
    total_synced = 0

    async with httpx.AsyncClient() as client:
        while True:
            url = f"{REDMINE_URL}/issues.json?limit={limit}&offset={offset}&status_id=*"
            resp = await client.get(url, headers=headers)
            data = resp.json()
            issues = data.get("issues", [])
            if not issues:
                break
            for i in issues:
                text = f"{i.get('subject', '')}\n{i.get('description', '')}"
                embedding = get_embedding(text)
                store_issue(Issue(
                    id=i["id"],
                    title=i.get("subject", ""),
                    description=i.get("description", "") or ""
                ), embedding)
                total_synced += 1
            offset += limit

    return {"status": "ok", "synced": total_synced}


@app.get("/health")
def health():
    return {"status": "ok"}
